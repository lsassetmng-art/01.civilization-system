   633	/**
   634	 * IMPORTANT:
   635	 * Do not pass SQL using `psql -c` here.
   636	 * In the previous version, psql variables like :'app_surface_code'
   637	 * were not substituted and were sent to PostgreSQL as-is.
   638	 *
   639	 * This fixed version passes SQL via stdin.
   640	 */
   641	function psqlJson(sql, vars = {}) {
   642	  const args = [
   643	    databaseUrl,
   644	    "-X",
   645	    "-A",
   646	    "-t",
   647	    "-v",
   648	    "ON_ERROR_STOP=1"
   649	  ];
   650	
   651	  for (const [key, value] of Object.entries(vars)) {
   652	    args.push("-v", `${key}=${value === null || value === undefined ? "" : String(value)}`);
   653	  }
   654	
   655	  const result = spawnSync("psql", args, {
   656	    input: sql,
   657	    encoding: "utf8",
   658	    maxBuffer: 1024 * 1024 * 20
   659	  });
   660	
   661	  if (result.status !== 0) {
   662	    const message = result.stderr || result.stdout || "psql failed";
   663	    const err = new Error(message.trim());
   664	    err.stderr = result.stderr;
   665	    err.stdout = result.stdout;
   666	    err.status = result.status;
   667	    throw err;
   668	  }
   669	
   670	  const text = (result.stdout || "").trim();
   671	  if (!text) return null;
   672	
   673	  try {
   674	    return JSON.parse(text);
   675	  } catch (e) {
   676	    throw new Error(`Failed to parse DB JSON: ${text}`);
   677	  }
   678	}
